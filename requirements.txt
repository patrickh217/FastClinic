# FastClinic runtime dependencies.
#
# There is no database driver here on purpose: MedBackend is the system of record.
# If psycopg2, sqlalchemy or an ORM ever reappears in this file, something has gone
# wrong architecturally - see ADR-001.

python-fasthtml          # FastHTML + HTMX; bundles htmx itself, do not add a second copy
uvicorn                  # ASGI server
httpx[http2]             # the only HTTP client; one pooled instance in services/medbackend
python-dotenv            # .env loading for local development
python-dateutil          # FHIR date parsing
PyYAML                   # SEO prompt/config files
markdownify              # user-guide rendering
openpyxl                 # .xlsx export buttons

# AI copilot - LangGraph agent with a configurable provider. xAI (Grok) and OpenAI
# both run through langchain-openai; xAI is API-compatible.
langgraph
langchain-core
langchain-openai
